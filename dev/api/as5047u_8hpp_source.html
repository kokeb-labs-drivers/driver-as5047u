28// Port hooks: a `Declares*` concept only tests the name, the `Has*` concept the exact signature,

152constexpr NonVolatileRegisters EncodeSettings(const Config& c,
153 const NonVolatileRegisters& now) noexcept {
154 using reg::At;
155 using reg::NvIndex;
156 NonVolatileRegisters r = now;
157 const auto zero = static_cast<uint32_t>(c.zero_position);
158 r[At(NvIndex::Disable)] = static_cast<uint16_t>(
159 BitIf(!c.uvw_enable, reg::kDisableUvwOff) | BitIf(!c.abi_enable, reg::kDisableAbiOff) |
160 BitIf(!c.adaptive_filter, reg::kDisableFilterOff));
161 r[At(NvIndex::Zposm)] = static_cast<uint16_t>((zero >> reg::kZeroHighShift) & reg::kZposmMask);
162 r[At(NvIndex::Zposl)] =
163 static_cast<uint16_t>((static_cast<uint32_t>(now[At(NvIndex::Zposl)]) & reg::kZposlDiaBits) |
164 (zero & reg::kZposlMask));
165 r[At(NvIndex::Settings1)] = static_cast<uint16_t>(
166 (static_cast<uint32_t>(now[At(NvIndex::Settings1)]) & reg::kS1DiaBits) |
167 ((static_cast<uint32_t>(c.filter_k_min_code) << reg::kS1KminShift) & reg::kS1KminMask) |
168 (static_cast<uint32_t>(c.filter_k_max_code) & reg::kS1KmaxMask));
169 r[At(NvIndex::Settings2)] = static_cast<uint16_t>(
170 BitIf(c.index_width == IndexWidth::OneLsb, reg::kS2IndexWidth1) |
171 BitIf(c.high_temperature_noise, reg::kS2NoiseSet) |
172 BitIf(c.direction == Direction::CounterClockwise, reg::kS2Dir) |
173 BitIf(c.incremental == IncrementalOutput::Uvw, reg::kS2UvwAbi) |
174 BitIf(!c.dynamic_compensation, reg::kS2DaecDis) | BitIf(c.abi_decimal, reg::kS2AbiDecimal) |
175 BitIf(c.pwm_enable, reg::kS2PwmOn) |
176 BitIf(c.uncompensated_at_angle_register, reg::kS2DataSelect));
177 r[At(NvIndex::Settings3)] = static_cast<uint16_t>(
178 ((static_cast<uint32_t>(c.uvw_pole_pairs) - 1U) & reg::kS3UvwppMask) |
179 ((static_cast<uint32_t>(c.hysteresis_code) << reg::kS3HysShift) & reg::kS3HysMask) |
180 ((static_cast<uint32_t>(c.abi_resolution_code) << reg::kS3AbiresShift) & reg::kS3AbiresMask));
181 return r;
182}

186constexpr Config DecodeSettings(const NonVolatileRegisters& r, const Config& base) noexcept {
187 using reg::At;
188 using reg::NvIndex;
189 Config c = base;
190 const auto disable = static_cast<uint32_t>(r[At(NvIndex::Disable)]);
191 c.uvw_enable = (disable & reg::kDisableUvwOff) == 0U;
192 c.abi_enable = (disable & reg::kDisableAbiOff) == 0U;
193 c.adaptive_filter = (disable & reg::kDisableFilterOff) == 0U;
194 c.zero_position = static_cast<uint16_t>(
195 ((static_cast<uint32_t>(r[At(NvIndex::Zposm)]) & reg::kZposmMask) << reg::kZeroHighShift) |
196 (static_cast<uint32_t>(r[At(NvIndex::Zposl)]) & reg::kZposlMask));
197 const auto s1 = static_cast<uint32_t>(r[At(NvIndex::Settings1)]);
198 c.filter_k_min_code = static_cast<uint8_t>((s1 & reg::kS1KminMask) >> reg::kS1KminShift);
199 c.filter_k_max_code = static_cast<uint8_t>(s1 & reg::kS1KmaxMask);
200 const auto s2 = static_cast<uint32_t>(r[At(NvIndex::Settings2)]);
201 c.index_width = ((s2 & reg::kS2IndexWidth1) != 0U) ? IndexWidth::OneLsb : IndexWidth::ThreeLsb;
202 c.high_temperature_noise = (s2 & reg::kS2NoiseSet) != 0U;
203 c.direction = ((s2 & reg::kS2Dir) != 0U) ? Direction::CounterClockwise : Direction::Clockwise;
204 c.incremental = ((s2 & reg::kS2UvwAbi) != 0U) ? IncrementalOutput::Uvw : IncrementalOutput::Abi;
205 c.dynamic_compensation = (s2 & reg::kS2DaecDis) == 0U;
206 c.abi_decimal = (s2 & reg::kS2AbiDecimal) != 0U;
207 c.uncompensated_at_angle_register = (s2 & reg::kS2DataSelect) != 0U;
208 c.pwm_enable = (s2 & reg::kS2PwmOn) != 0U;
209 const auto s3 = static_cast<uint32_t>(r[At(NvIndex::Settings3)]);
210 const uint32_t pole_pairs = (s3 & reg::kS3UvwppMask) + 1U;
211 // UVWPP code 7 also means 7 pole pairs (Figure 48).
212 c.uvw_pole_pairs =
213 static_cast<uint8_t>((pole_pairs > kMaxUvwPolePairs) ? kMaxUvwPolePairs : pole_pairs);
214 c.hysteresis_code = static_cast<uint8_t>((s3 & reg::kS3HysMask) >> reg::kS3HysShift);
215 c.abi_resolution_code = static_cast<uint8_t>((s3 & reg::kS3AbiresMask) >> reg::kS3AbiresShift);
216 return c;
217}

626 template <typename T>
627 Result<T> afterAccess(const Result<T>& r) noexcept {
628 if (!reset_hint_ && !lost_hint_) {
629 return r;
630 }
631 if (!initialized_) {
632 reset_hint_ = false; // already uninitialised: EnsureInitialized() writes everything
633 lost_hint_ = false;
634 return r;
635 }
636 // A reset replaces the call's own result. A compare that could not be read leaves a failed
637 // call's own error, but fails a call that succeeded with the compare's error: no value is
638 // returned while a reset may have gone unseen (the compare stays pending).
639 const auto c = checkReset();
640 if (c.error == Error::DeviceReset) {
641 return Result<T>::Failure(Error::DeviceReset);
642 }
643 if (!c && r.ok()) {
644 return Result<T>::Failure(c.error);
645 }
646 return r;
647 }
