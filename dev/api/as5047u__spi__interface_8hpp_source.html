113constexpr uint32_t Crc8Step(uint32_t crc, uint32_t byte) noexcept {
114 uint32_t c = (crc ^ byte) & 0xFFU;
115 for (uint32_t bit = 0U; bit < 8U; ++bit) {
116 c = ((c & 0x80U) != 0U) ? (((c << 1U) ^ reg::kCrcPolynomial) & 0xFFU) : ((c << 1U) & 0xFFU);
117 }
118 return c;
119}

122constexpr uint8_t Crc8Bitwise(uint16_t word) noexcept {
123 const uint32_t high =
124 Crc8Step(static_cast<uint32_t>(reg::kCrcInit), static_cast<uint32_t>(word) >> 8U);
125 const uint32_t low = Crc8Step(high, static_cast<uint32_t>(word) & 0xFFU);
126 return static_cast<uint8_t>(low ^ reg::kCrcFinalXor);
127}

151constexpr uint8_t Crc8(uint16_t word) noexcept {
152 const auto high = static_cast<uint32_t>(detail::kCrc8Table[static_cast<std::size_t>(
153 (reg::kCrcInit ^ (static_cast<uint32_t>(word) >> 8U)) & 0xFFU)]);
154 const auto low = static_cast<uint32_t>(
155 detail::kCrc8Table[static_cast<std::size_t>((high ^ static_cast<uint32_t>(word)) & 0xFFU)]);
156 return static_cast<uint8_t>(low ^ reg::kCrcFinalXor);
157}

191constexpr bool IsLinePattern(uint16_t reply, uint8_t crc) noexcept {
192 return ((reply == 0x0000U) && (crc == 0x00U)) || ((reply == 0xFFFFU) && (crc == 0xFFU));

219 Result<uint16_t> Exchange(FrameFormat format, uint8_t pad, uint16_t word) noexcept {
220 if (resync_) {
221 (void)transfer(format, pad, kResyncWord); // its reply is the previous command's; unused
222 if (resync_) {
223 return Result<uint16_t>::Failure(Error::Bus); // still failing: send nothing else
224 }
225 }
226 return transfer(format, pad, word);
227 }

231 return resync_;
232 }

237 return last_reply_;
238 }

242 static constexpr uint16_t kResyncWord = 0x4000U;

244 Result<uint16_t> transfer(FrameFormat format, uint8_t pad, uint16_t word) noexcept {
245 std::array<uint8_t, kMaxFrameBytes> tx{};
246 std::array<uint8_t, kMaxFrameBytes> rx{};
247 const std::size_t len = FrameBytes(format);
248 std::size_t at = 0U;
249 if (format == FrameFormat::Bits32) {
250 tx[0] = pad;
251 at = 1U;
252 }
253 tx[at] = static_cast<uint8_t>(static_cast<uint32_t>(word) >> 8U);
254 tx[at + 1U] = static_cast<uint8_t>(static_cast<uint32_t>(word) & 0xFFU);
255 if (format != FrameFormat::Bits16) {
256 tx[at + 2U] = Crc8(word);
257 }
258 resync_ = !port_.Transfer(tx.data(), rx.data(), len);
259 if (resync_) {
260 return Result<uint16_t>::Failure(Error::Bus);
261 }
262 const auto reply =
263 static_cast<uint16_t>((static_cast<uint32_t>(rx[0]) << 8U) | static_cast<uint32_t>(rx[1]));
264 last_reply_ = reply;
265 if ((format != FrameFormat::Bits16) && (rx[2] != Crc8(reply))) {
266 return Result<uint16_t>::Failure(IsLinePattern(reply, rx[2]) ? Error::NoDevice : Error::Crc);
267 }
268 return Result<uint16_t>::Success(reply);
269 }
270
271 Port& port_;
272 uint16_t last_reply_ = 0;
273 bool resync_ = false;
274};
