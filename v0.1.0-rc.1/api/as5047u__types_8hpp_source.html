422constexpr Error Validate(const Config& config) noexcept {
423 if (!detail::IsEnumerator(config.frame) || !detail::IsEnumerator(config.direction) ||
424 !detail::IsEnumerator(config.incremental) || !detail::IsEnumerator(config.index_width)) {
425 return Error::InvalidArgument;
426 }
427 if ((config.zero_position > kMaxAngleCount) || (config.filter_k_min_code > kMax3BitCode) ||
428 (config.filter_k_max_code > kMax3BitCode) || (config.abi_resolution_code > kMax3BitCode) ||
429 (config.hysteresis_code > kMaxHysteresisCode) || (config.uvw_pole_pairs == 0U) ||
430 (config.uvw_pole_pairs > kMaxUvwPolePairs) ||
431 (config.startup_timeout_ms > kStartupTimeoutMaxMs)) {
432 return Error::InvalidArgument;
433 }
434 return Error::Ok;
435}

437static_assert(Validate(Config{}) == Error::Ok, "the defaults are valid");
